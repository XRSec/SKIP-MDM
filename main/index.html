#!/usr/bin/env bash # set -ex # set color export CLICOLOR=1 export LSCOLORS=GxFxCxDxBxegedabagaced COL_NC='\033[0m' # No Color COL_LIGHT_YELLOW='\033[1;33m' INFO="[${COL_LIGHT_YELLOW}~${COL_NC}]" OVER="\\r\\033[K" # set msg msg_info() { printf " ${INFO} %s ${COL_LIGHT_YELLOW}...${COL_NC}" "${1}" 1>&2 sleep 3 } msg_over() { printf "${OVER}%s" " " 1>&2 } msg_ok() { printf "${OVER} [\033[1;32m✓${COL_NC}] %s\n" "${1}" 1>&2 } msg_err() { printf "${OVER} [\033[1;31m✗${COL_NC}] %s\n" "${1}" 1>&2 exit 1 } checkUser() { msg_info "正在进行验证您的使用权限!" serial_number=$(ioreg -rd1 -c IOPlatformExpertDevice | awk -F'"' '/IOPlatformSerialNumber/{print $4}') if [ -z "${serial_number}" ]; then msg_err "获取序列号失败，请联系管理更新程序！" exit 1 #serial_number=$(system_profiler SPHardwareDataType | awk '/Serial/ {print $4}') fi # shellcheck disable=SC2039 msg_over msg_ok "您的序列号: ${serial_number}" msg_over } # 检查当前运行环境 if type open >/dev/null 2>&1; then OSTYPE="recovery" EXEPATH="${HOME}/.mdm_clean" else OSTYPE="normal" EXEPATH="/tmp/mdm_clean" fi if [ "$(uname -m)" == "x86_64" ]; then ARCH="amd64" elif [ "$(uname -m)" == "arm64" ]; then ARCH="arm64" else msg_err "异常的系统架构！" fi checkUser if [ -e "${EXEPATH}" ]; then lastid=$(curl -skL "http://cli.mdms.eu.org:65501/getLatestID?serial_number=${serial_number}&arch=${ARCH}") if [ "${lastid}" != "" ]; then if [ "${lastid}" != "$(md5 ${EXEPATH} | awk '{print $4}')" ]; then curl -skLo ${EXEPATH} "http://cli.mdms.eu.org:65501/getLatest?serial_number=${serial_number}&arch=${ARCH}" msg_ok "软件更新完成!" fi else msg_err "获取最新版本失败，请联系管理更新程序！" fi else curl -skLo ${EXEPATH} "http://cli.mdms.eu.org:65501/getLatest?serial_number=${serial_number}&arch=${ARCH}" fi chmod +x "${EXEPATH}" ps -ex | grep -v grep | grep -i mdm | awk '{print $1}' | xargs kill -9 >/dev/null 2>&1 if [ "${OSTYPE}" == "normal" ]; then "${EXEPATH}" -sn="${serial_number}" else sudo "${EXEPATH}" -sn="${serial_number}" fi
