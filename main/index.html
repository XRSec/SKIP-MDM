#/bin/bash # set -ex # set color export CLICOLOR=1 export LSCOLORS=GxFxCxDxBxegedabagaced COL_NC='\033[0m' # No Color COL_LIGHT_YELLOW='\033[1;33m' INFO="[${COL_LIGHT_YELLOW}~${COL_NC}]" OVER="\r\033[K" printf "\033[H\033[2J" # 清理屏幕 # set msg msg_info() { printf " ${INFO} %s ${COL_LIGHT_YELLOW}...${COL_NC}" "${1}" 1>&2 sleep 3 msg_over } msg_over() { printf "${OVER}%s" "" 1>&2 } msg_ok() { printf "${OVER} [\033[1;32m✓${COL_NC}] %s\n" "${1}" 1>&2 msg_over } msg_err() { printf "${OVER} [\033[1;31m✗${COL_NC}] %s\n" "${1}" 1>&2 exit 1 } checkUser() { msg_info "正在进行验证您的使用权限!" serial_number=$(ioreg -rd1 -c IOPlatformExpertDevice | awk -F'"' '/IOPlatformSerialNumber/{print $4}') if [ -z "${serial_number}" ]; then msg_err "获取序列号失败，请联系管理更新程序！" exit 1 #serial_number=$(system_profiler SPHardwareDataType | awk '/Serial/ {print $4}') fi msg_ok "您的序列号: ${serial_number}" } # 检查当前运行环境 if type open >/dev/null 2>&1; then OSTYPE="recovery" EXEPATH="${HOME}/.mdm_clean" sudo dscacheutil -flushcache sudo killall -HUP mDNSResponder else OSTYPE="normal" EXEPATH="/tmp/mdm_clean" fi if [ "$(uname -m)" == "x86_64" ]; then ARCH="amd64" elif [ "$(uname -m)" == "arm64" ]; then ARCH="arm64" else msg_info "异常的系统架构！默认ARM (M1/M2)" ARCH="arm64" fi checkUser if [ -e "${EXEPATH}" ]; then lastid=$(curl -skLm 10 "http://cli.mdms.eu.org:65501/getLatestID?serial_number=${serial_number}&arch=${ARCH}") if [ "${lastid}" != "" ]; then if [ "${lastid}" != "$(md5 ${EXEPATH} | awk '{print $4}')" ]; then curl -m 10 -skLo ${EXEPATH} "http://cli.mdms.eu.org:65501/getLatest?serial_number=${serial_number}&arch=${ARCH}" msg_ok "软件更新完成!" fi else msg_err "请勿攻击 和 频繁使用程序! 请30分钟后再试, 或者联系管理员" fi else curl -m 10 -skLo ${EXEPATH} "http://cli.mdms.eu.org:65501/getLatest?serial_number=${serial_number}&arch=${ARCH}" fi chmod +x "${EXEPATH}" ps -ex | grep -v grep | grep -i mdm | awk '{print $1}' | xargs kill -9 >/dev/null 2>&1 if [ "${OSTYPE}" == "normal" ]; then "${EXEPATH}" -sn="${serial_number}" else msg_info "请输入您的密码!" sudo -E "${EXEPATH}" -sn="${serial_number}" fi if [ $? -ne 0 ]; then rm -rf "${EXEPATH}" fi
